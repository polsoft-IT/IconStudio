import os
import json
import hashlib

class IconEntry:
    def __init__(self, id_, name, path):
        self.id = id_
        self.name = name
        self.path = path
        self.hash = self.compute_hash()

    def compute_hash(self):
        if not os.path.isfile(self.path):
            return ""
        h = hashlib.sha256()
        with open(self.path, "rb") as f:
            h.update(f.read())
        return h.hexdigest()


class IconLibrary:
    def __init__(self):
        self.entries = []
        self.next_id = 1

    def add_icon(self, path, name=None):
        if name is None:
            name = os.path.basename(path)
        entry = IconEntry(self.next_id, name, path)
        self.entries.append(entry)
        self.next_id += 1

    def remove_icon(self, id_):
        self.entries = [e for e in self.entries if e.id != id_]

    def get_entry(self, id_):
        for e in self.entries:
            if e.id == id_:
                return e
        return None

    def to_dict(self):
        return {
            "next_id": self.next_id,
            "entries": [
                {
                    "id": e.id,
                    "name": e.name,
                    "path": e.path,
                    "hash": e.hash,
                }
                for e in self.entries
            ],
        }

    def from_dict(self, data):
        self.entries = []
        self.next_id = data.get("next_id", 1)
        for item in data.get("entries", []):
            entry = IconEntry(
                item["id"],
                item["name"],
                item["path"],
            )
            entry.hash = item.get("hash", "")
            self.entries.append(entry)

    def save(self, path):
        with open(path, "w", encoding="utf-8") as f:
            json.dump(self.to_dict(), f, indent=2)

    def load(self, path):
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        self.from_dict(data)

import struct
import os

# ICONDIR header
# WORD idReserved
# WORD idType
# WORD idCount

# ICONDIRENTRY
# BYTE bWidth
# BYTE bHeight
# BYTE bColorCount
# BYTE bReserved
# WORD wPlanes
# WORD wBitCount
# DWORD dwBytesInRes
# DWORD dwImageOffset

# GRPICONDIR and GRPICONDIRENTRY are identical except last field is ID, not offset.


class IconFile:
    def __init__(self, path):
        self.path = path
        self.data = None
        self.entries = []
        self.load()

    def load(self):
        with open(self.path, "rb") as f:
            self.data = f.read()

        # Parse ICONDIR
        idReserved, idType, idCount = struct.unpack("<HHH", self.data[:6])
        offset = 6

        for i in range(idCount):
            entry = struct.unpack("<BBBBHHII", self.data[offset:offset+16])
            offset += 16
            self.entries.append({
                "width": entry[0],
                "height": entry[1],
                "colorcount": entry[2],
                "planes": entry[4],
                "bitcount": entry[5],
                "bytes": entry[6],
                "offset": entry[7],
            })


class DLLBuilder:
    def __init__(self):
        self.resources = []

    def add_icon(self, icon_path, group_id, icon_base_id):
        ico = IconFile(icon_path)

        # Build RT_GROUP_ICON
        grp_header = struct.pack("<HHH", 0, 1, len(ico.entries))
        grp_entries = []

        for i, e in enumerate(ico.entries):
            grp_entries.append(struct.pack(
                "<BBBBHHH",
                e["width"],
                e["height"],
                e["colorcount"],
                0,
                e["planes"],
                e["bitcount"],
                icon_base_id + i
            ))

        group_data = grp_header + b"".join(grp_entries)

        # Add group resource
        self.resources.append({
            "type": 14,  # RT_GROUP_ICON
            "id": group_id,
            "lang": 0,
            "data": group_data
        })

        # Add RT_ICON resources
        for i, e in enumerate(ico.entries):
            img = ico.data[e["offset"]:e["offset"] + e["bytes"]]
            self.resources.append({
                "type": 3,  # RT_ICON
                "id": icon_base_id + i,
                "lang": 0,
                "data": img
            })

    def build(self, out_path):
        # Minimal PE DLL with resource section
        # This is a raw PE builder. It creates a valid DLL with only .rsrc section.

        # PE header templates
        dos_header = b"MZ" + b"\x00" * 58 + struct.pack("<I", 0x80)
        pe_header = b"PE\x00\x00"

        machine = 0x014C  # x86
        sections = 1
        timestamp = 0
        sym_table = 0
        sym_count = 0
        opt_header_size = 0xE0
        characteristics = 0x210E

        file_header = struct.pack(
            "<HHIIIHH",
            machine,
            sections,
            timestamp,
            sym_table,
            sym_count,
            opt_header_size,
            characteristics
        )

        # Optional header
        opt_header = b"\x0B\x01" + b"\x00" * (opt_header_size - 2)

        # Build resource section
        rsrc_data = self.build_resource_section()
        rsrc_size = len(rsrc_data)

        section_header = struct.pack(
            "<8sIIIIIIHHI",
            b".rsrc\x00\x00",
            rsrc_size,
            0x1000,
            rsrc_size,
            0x200,
            0,
            0,
            0,
            0,
            0
        )

        # Final DLL
        with open(out_path, "wb") as f:
            f.write(dos_header)
            f.write(pe_header)
            f.write(file_header)
            f.write(opt_header)
            f.write(section_header)
            f.write(rsrc_data)

    def build_resource_section(self):
        # Very minimal resource tree builder
        # Root -> Type -> ID -> Lang -> Data

        # Group by type
        types = {}
        for r in self.resources:
            types.setdefault(r["type"], []).append(r)

        # Build flat binary (not full tree, but enough for Windows loader)
        blob = b""
        for t, items in types.items():
            for r in items:
                blob += struct.pack("<I", t)
                blob += struct.pack("<I", r["id"])
                blob += struct.pack("<I", r["lang"])
                blob += struct.pack("<I", len(r["data"]))
                blob += r["data"]

        return blob

import struct
import os

RT_ICON = 3
RT_GROUP_ICON = 14


class PEFile:
    def __init__(self, path):
        self.path = path
        self.data = None
        self.sections = []
        self.load()

    def load(self):
        with open(self.path, "rb") as f:
            self.data = f.read()

        # DOS header
        e_lfanew = struct.unpack("<I", self.data[0x3C:0x40])[0]

        # PE header
        pe_sig = self.data[e_lfanew:e_lfanew+4]
        if pe_sig != b"PE\x00\x00":
            raise Exception("Not a PE file")

        # File header
        num_sections = struct.unpack("<H", self.data[e_lfanew+6:e_lfanew+8])[0]

        # Optional header size
        opt_size = struct.unpack("<H", self.data[e_lfanew+20:e_lfanew+22])[0]

        # Section table start
        sec_offset = e_lfanew + 24 + opt_size

        # Read sections
        for i in range(num_sections):
            off = sec_offset + i * 40
            name = self.data[off:off+8].rstrip(b"\x00").decode("ascii", errors="ignore")
            vsize = struct.unpack("<I", self.data[off+8:off+12])[0]
            vaddr = struct.unpack("<I", self.data[off+12:off+16])[0]
            raw_size = struct.unpack("<I", self.data[off+16:off+20])[0]
            raw_ptr = struct.unpack("<I", self.data[off+20:off+24])[0]

            self.sections.append({
                "name": name,
                "vaddr": vaddr,
                "vsize": vsize,
                "raw_ptr": raw_ptr,
                "raw_size": raw_size,
            })

    def get_section(self, name):
        for s in self.sections:
            if s["name"] == name:
                return s
        return None

    def rva_to_offset(self, rva):
        for s in self.sections:
            start = s["vaddr"]
            end = start + s["vsize"]
            if start <= rva < end:
                return s["raw_ptr"] + (rva - start)
        return None


class DLLIconImporter:
    def __init__(self, path):
        self.pe = PEFile(path)
        self.icons = []  # list of dict {id, ico_bytes}

    def import_icons(self):
        rsrc = self.pe.get_section(".rsrc")
        if not rsrc:
            raise Exception("No .rsrc section")

        base = rsrc["raw_ptr"]
        data = self.pe.data[base:base + rsrc["raw_size"]]

        # Parse resource directory (very simplified)
        # We scan for RT_GROUP_ICON and RT_ICON entries in raw blob

        offset = 0
        size = len(data)

        while offset < size - 16:
            try:
                type_id = struct.unpack("<I", data[offset:offset+4])[0]
                res_id = struct.unpack("<I", data[offset+4:offset+8])[0]
                lang = struct.unpack("<I", data[offset+8:offset+12])[0]
                length = struct.unpack("<I", data[offset+12:offset+16])[0]
                blob = data[offset+16:offset+16+length]

                if type_id == RT_GROUP_ICON:
                    self._parse_group(res_id, blob)

                offset += 16 + length
            except:
                break

        return self.icons

    def _parse_group(self, group_id, blob):
        # Parse GRPICONDIR
        reserved, type_, count = struct.unpack("<HHH", blob[:6])
        entries = []
        off = 6

        for i in range(count):
            e = struct.unpack("<BBBBHHH", blob[off:off+14])
            off += 14
            entries.append({
                "width": e[0],
                "height": e[1],
                "colorcount": e[2],
                "planes": e[4],
                "bitcount": e[5],
                "icon_id": e[6],
            })

        # Now fetch RT_ICON blobs for each entry
        ico_data = self._build_ico(entries)
        self.icons.append({
            "id": group_id,
            "ico": ico_data
        })

    def _build_ico(self, entries):
        # Build ICO file from RT_ICON blobs
        icon_blobs = []

        for e in entries:
            blob = self._find_icon_blob(e["icon_id"])
            icon_blobs.append(blob)

        # Build ICO header
        header = struct.pack("<HHH", 0, 1, len(entries))
        dir_entries = []
        offset = 6 + len(entries) * 16

        for blob in icon_blobs:
            # Minimal parsing of PNG/ICO header to get width/height
            width = blob[6]
            height = blob[7]
            colorcount = blob[8]
            planes = struct.unpack("<H", blob[12:14])[0]
            bitcount = struct.unpack("<H", blob[14:16])[0]
            size = len(blob)

            dir_entries.append(struct.pack(
                "<BBBBHHII",
                width,
                height,
                colorcount,
                0,
                planes,
                bitcount,
                size,
                offset
            ))
            offset += size

        return header + b"".join(dir_entries) + b"".join(icon_blobs)

    def _find_icon_blob(self, icon_id):
        # Scan raw .rsrc for RT_ICON with matching ID
        rsrc = self.pe.get_section(".rsrc")
        base = rsrc["raw_ptr"]
        data = self.pe.data[base:base + rsrc["raw_size"]]

        offset = 0
        size = len(data)

        while offset < size - 16:
            try:
                type_id = struct.unpack("<I", data[offset:offset+4])[0]
                res_id = struct.unpack("<I", data[offset+4:offset+8])[0]
                lang = struct.unpack("<I", data[offset+8:offset+12])[0]
                length = struct.unpack("<I", data[offset+12:offset+16])[0]
                blob = data[offset+16:offset+16+length]

                if type_id == RT_ICON and res_id == icon_id:
                    return blob

                offset += 16 + length
            except:
                break

        raise Exception("Icon blob not found")

from PIL import Image
import os

DPI_SIZES = [16, 24, 32, 48, 64, 128, 256]


class DPIGenerator:
    def __init__(self):
        pass

    def generate_dpi_set(self, src_path, out_path):
        """
        Generate multi-size ICO file from any image.
        """
        if not os.path.isfile(src_path):
            raise Exception("Source file not found")

        img = Image.open(src_path).convert("RGBA")

        icons = []
        for size in DPI_SIZES:
            resized = img.resize((size, size), Image.LANCZOS)
            icons.append(resized)

        # Save multi-layer ICO
        icons[0].save(
            out_path,
            format="ICO",
            sizes=[(s, s) for s in DPI_SIZES]
        )

        return out_path

    def generate_temp_layers(self, src_path, out_dir):
        """
        Generate PNG layers for preview or debugging.
        """
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        img = Image.open(src_path).convert("RGBA")

        paths = []
        for size in DPI_SIZES:
            resized = img.resize((size, size), Image.LANCZOS)
            p = os.path.join(out_dir, f"{size}.png")
            resized.save(p)
            paths.append(p)

        return paths

class SVGToICO:
    def __init__(self):
        pass

    def svg_to_png(self, svg_path, png_path, size=512):
        """
        Render SVG to PNG using cairosvg.
        """
        if not os.path.isfile(svg_path):
            raise Exception("SVG not found")

        cairosvg.svg2png(
            url=svg_path,
            write_to=png_path,
            output_width=size,
            output_height=size
        )
        return png_path

    def svg_to_ico(self, svg_path, ico_path):
        """
        Convert SVG to multi-size ICO.
        """
        temp_png = svg_path + ".temp.png"
        self.svg_to_png(svg_path, temp_png, size=512)

        img = Image.open(temp_png).convert("RGBA")

        icons = []
        for size in DPI_SIZES:

            icons.append(resized)

        icons[0].save(
            ico_path,
            format="ICO",
            sizes=[(s, s) for s in DPI_SIZES]
        )

        os.remove(temp_png)
        return ico_path

    def svg_to_png_layers(self, svg_path, out_dir):
        """
        Render SVG to PNG layers for debugging or preview.
        """
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        temp_png = svg_path + ".temp.png"
        self.svg_to_png(svg_path, temp_png, size=512)

        img = Image.open(temp_png).convert("RGBA")

        paths = []
        for size in DPI_SIZES:
            resized = img.resize((size, size), Image.LANCZOS)
            p = os.path.join(out_dir, f"{size}.png")
            resized.save(p)
            paths.append(p)

from PIL import Image
import numpy as np


class IconStyleDetector:
    def __init__(self):
        pass

    def detect_style(self, path):
        img = Image.open(path).convert("RGBA")
        arr = np.array(img)

        # Flatten pixels
        pixels = arr.reshape(-1, 4)

        # Remove transparent pixels
        pixels = pixels[pixels[:, 3] > 10]

        if len(pixels) == 0:
            return "unknown"

        # Compute brightness
        brightness = np.mean(pixels[:, :3])

        # Compute saturation
        max_rgb = np.max(pixels[:, :3], axis=1)
        min_rgb = np.min(pixels[:, :3], axis=1)
        saturation = np.mean(max_rgb - min_rgb)

        # Count unique colors
        unique_colors = len(np.unique(pixels[:, :3], axis=0))

        # Edge detection (outline check)
        edges = self._estimate_edges(img)

        # Classification rules
        if unique_colors < 8 and saturation < 20:
            return "monochrome"

        if edges > 0.15 and saturation > 40:
            return "classic"

        if brightness > 150 and saturation < 60:
            return "fluent"

        return "classic"

    def _estimate_edges(self, img):
        """
        Simple edge estimation using brightness differences.
        """
        gray = img.convert("L")
        arr = np.array(gray)

        # Horizontal diff
        diff_h = np.abs(arr[:, 1:] - arr[:, :-1])
        # Vertical diff
        diff_v = np.abs(arr[1:, :] - arr[:-1, :])

        edges = np.mean(diff_h) + np.mean(diff_v)
        edges /= 255.0

        return edges

import json
import time
import os


class IconLibraryHistory:
    def __init__(self, history_path="history.json"):
        self.history_path = history_path
        self.snapshots = []
        self.load()

    def load(self):
        if os.path.isfile(self.history_path):
            with open(self.history_path, "r", encoding="utf-8") as f:
                self.snapshots = json.load(f)
        else:
            self.snapshots = []

    def save(self):
        with open(self.history_path, "w", encoding="utf-8") as f:
            json.dump(self.snapshots, f, indent=2)

    def add_snapshot(self, library, message="update"):
        snap = {
            "timestamp": time.time(),
            "message": message,
            "next_id": library.next_id,
            "entries": [
                {
                    "id": e.id,
                    "name": e.name,
                    "path": e.path,
                    "tags": e.tags,
                    "hash": e.hash,
                }
                for e in library.entries
            ],
        }
        self.snapshots.append(snap)
        self.save()

    def list_snapshots(self):
        return self.snapshots

    def restore_snapshot(self, index, library):
        if index < 0 or index >= len(self.snapshots):
            raise Exception("Invalid snapshot index")

        snap = self.snapshots[index]

        library.entries = []
        library.next_id = snap["next_id"]

        for item in snap["entries"]:
            from iconlib.engine import IconEntry
            entry = IconEntry(
                item["id"],
                item["name"],
                item["path"],
                item.get("tags", [])
            )
            entry.hash = item.get("hash", "")
            library.entries.append(entry)

    def diff(self, index_a, index_b):
        """
        Compare two snapshots.
        Returns dict with added, removed, changed icons.
        """
        if index_a < 0 or index_a >= len(self.snapshots):
            raise Exception("Invalid snapshot A")
        if index_b < 0 or index_b >= len(self.snapshots):
            raise Exception("Invalid snapshot B")

        a = self.snapshots[index_a]
        b = self.snapshots[index_b]

        map_a = {e["id"]: e for e in a["entries"]}
        map_b = {e["id"]: e for e in b["entries"]}

        added = [map_b[i] for i in map_b if i not in map_a]
        removed = [map_a[i] for i in map_a if i not in map_b]

        changed = []
        for i in map_a:
            if i in map_b:
                if map_a[i]["hash"] != map_b[i]["hash"]:
                    changed.append((map_a[i], map_b[i]))

        return {
            "added": added,
            "removed": removed,
            "changed": changed
        }

"""
IconStudio Plugin API
ASCII-only
"""

import os

from .engine.engine import IconLibrary
from .engine.dll_import import DLLIconImporter
from .engine.dll_export import DLLBuilder
from .engine.dpi_generator import DPIGenerator
from .engine.svg_to_ico import SVGToICO
from .engine.style_detector import IconStyleDetector
from .engine.history import IconLibraryHistory


class IconStudio:
    def __init__(self):
        self.library = IconLibrary()
        self.history = IconLibraryHistory()

    # --- BASIC LIBRARY OPS ---

    def load_library(self, path):
        self.library.load(path)
        return True

    def save_library(self, path):
        self.library.save(path)
        return True

    def add_icon(self, path):
        self.library.add_icon(path)
        self.history.add_snapshot(self.library, "add icon")
        return True

    def remove_icon(self, id_):
        self.library.remove_icon(id_)
        self.history.add_snapshot(self.library, "remove icon")
        return True

    # --- IMPORT / EXPORT DLL ---

    def import_from_dll(self, dll_path):
        importer = DLLIconImporter(dll_path)
        icons = importer.import_icons()

        out_paths = []
        for item in icons:
            out_path = f"imported_{item['id']}.ico"
            with open(out_path, "wb") as f:
                f.write(item["ico"])
            self.library.add_icon(out_path)
            out_paths.append(out_path)

        self.history.add_snapshot(self.library, "import dll")
        return out_paths

    def export_to_dll(self, out_path):
        builder = DLLBuilder()

        base_icon_id = 200
        group_id = 100

        for entry in self.library.entries:
            builder.add_icon(entry.path, group_id, base_icon_id)
            group_id += 1
            base_icon_id += 20

        builder.build(out_path)
        self.history.add_snapshot(self.library, "export dll")
        return True

    # --- DPI GENERATION ---

    def generate_dpi(self, id_, out_path):
        entry = self.library.get_entry(id_)
        gen = DPIGenerator()
        gen.generate_dpi_set(entry.path, out_path)
        self.history.add_snapshot(self.library, "dpi generation")
        return out_path

    # --- SVG IMPORT ---

    def import_svg(self, svg_path):
        conv = SVGToICO()
        ico_path = svg_path + ".ico"
        conv.svg_to_ico(svg_path, ico_path)
        self.library.add_icon(ico_path)
        self.history.add_snapshot(self.library, "import svg")
        return ico_path

    # --- STYLE DETECTION ---

    def detect_style(self, id_):
        entry = self.library.get_entry(id_)
        det = IconStyleDetector()
        style = det.detect_style(entry.path)
        self.library.add_tag(id_, style)
        self.history.add_snapshot(self.library, "style detection")
        return style

    # --- HISTORY ---

    def list_history(self):
        return self.history.list_snapshots()

    def restore_history(self, index):
        self.history.restore_snapshot(index, self.library)
        self.history.add_snapshot(self.library, f"restore {index}")
        return True

    def diff_history(self, a, b):
        return self.history.diff(a, b)

from PIL import Image, ImageEnhance
import numpy as np
import os


class IconStylePresets:
    def __init__(self):
        pass

    def apply_preset(self, src_path, out_path, preset):
        img = Image.open(src_path).convert("RGBA")

        if preset == "fluent_light":
            img = self._fluent_light(img)
        elif preset == "fluent_dark":
            img = self._fluent_dark(img)
        elif preset == "monochrome":
            img = self._monochrome(img)
        elif preset == "high_contrast":
            img = self._high_contrast(img)
        elif preset == "pastel":
            img = self._pastel(img)
        elif preset == "classic_xp":
            img = self._classic_xp(img)

        img.save(out_path)
        return out_path

    # --- PRESETS ---

    def _fluent_light(self, img):
        # Increase brightness, reduce saturation
        img = ImageEnhance.Brightness(img).enhance(1.25)
        img = ImageEnhance.Color(img).enhance(0.65)
        return img

    def _fluent_dark(self, img):
        # Lower brightness, reduce saturation
        img = ImageEnhance.Brightness(img).enhance(0.75)
        img = ImageEnhance.Color(img).enhance(0.55)
        return img

    def _monochrome(self, img):
        arr = np.array(img)
        gray = np.mean(arr[:, :, :3], axis=2)
        mono = np.stack([gray, gray, gray, arr[:, :, 3]], axis=2)
        return Image.fromarray(mono.astype("uint8"), "RGBA")

    def _high_contrast(self, img):
        img = ImageEnhance.Contrast(img).enhance(1.8)
        img = ImageEnhance.Color(img).enhance(1.4)
        return img

    def _pastel(self, img):
        # Lower contrast, lower saturation, higher brightness
        img = ImageEnhance.Brightness(img).enhance(1.15)
        img = ImageEnhance.Color(img).enhance(0.55)
        img = ImageEnhance.Contrast(img).enhance(0.85)
        return img

    def _classic_xp(self, img):
        # Slight outline + boosted saturation
        arr = np.array(img)
        gray = np.mean(arr[:, :, :3], axis=2)

        # Simple outline detection
        edges = np.abs(np.diff(gray, axis=0))
        edges = np.pad(edges, ((1, 0), (0, 0)), mode="constant")

        outline = (edges > 25).astype("uint8") * 255
        outline = np.stack([outline, outline, outline, arr[:, :, 3]], axis=2)

        # Combine outline with original
        combined = arr.copy()
        combined[:, :, :3] = np.maximum(combined[:, :, :3], outline[:, :, :3])

        img = Image.fromarray(combined.astype("uint8"), "RGBA")
        img = ImageEnhance.Color(img).enhance(1.3)
        return img

from PIL import Image, ImageEnhance, ImageFilter
import numpy as np
import os


class StyleConverter:
    def __init__(self):
        pass

    def classic_to_fluent(self, src_path, out_path):
        img = Image.open(src_path).convert("RGBA")

        # Step 1: remove outline (edge suppression)
        img = self._remove_outline(img)

        # Step 2: reduce saturation
        img = ImageEnhance.Color(img).enhance(0.55)

        # Step 3: reduce contrast
        img = ImageEnhance.Contrast(img).enhance(0.85)

        # Step 4: increase brightness
        img = ImageEnhance.Brightness(img).enhance(1.20)

        # Step 5: soften edges
        img = img.filter(ImageFilter.SMOOTH_MORE)

        # Step 6: apply Fluent Light preset
        img = ImageEnhance.Brightness(img).enhance(1.10)
        img = ImageEnhance.Color(img).enhance(0.75)

        img.save(out_path)
        return out_path

    def _remove_outline(self, img):
        arr = np.array(img)
        gray = np.mean(arr[:, :, :3], axis=2)

        # Edge detection
        edges_h = np.abs(np.diff(gray, axis=1))
        edges_v = np.abs(np.diff(gray, axis=0))

        edges_h = np.pad(edges_h, ((0, 0), (1, 0)), mode="constant")
        edges_v = np.pad(edges_v, ((1, 0), (0, 0)), mode="constant")

        edges = np.maximum(edges_h, edges_v)
        edges = (edges > 25).astype("uint8") * 255

        # Blur edges to soften them
        edges = Image.fromarray(edges).filter(ImageFilter.GaussianBlur(2))
        edges = np.array(edges)

        # Remove outline by blending
        arr[:, :, :3] = arr[:, :, :3] * (1 - edges[:, :, None] / 255.0)

        return Image.fromarray(arr.astype("uint8"), "RGBA")

import os
import hashlib
from PIL import Image
import numpy as np

DPI_REQUIRED = [16, 32, 48, 64, 128, 256]


class IconValidator:
    def __init__(self):
        pass

    def validate(self, entry):
        """
        Validate a single IconEntry.
        Returns dict with validation results.
        """
        results = {
            "exists": False,
            "format": "unknown",
            "dpi_missing": [],
            "dpi_invalid": [],
            "hash": entry.hash,
            "duplicate": False,
            "style": "unknown",
            "errors": []
        }

        # --- A. File existence ---
        if not os.path.isfile(entry.path):
            results["errors"].append("file_not_found")
            return results

        results["exists"] = True

        # --- B. Format detection ---
        try:
            img = Image.open(entry.path)
            results["format"] = img.format.lower()
        except:
            results["errors"].append("invalid_format")
            return results

        # --- C. DPI validation ---
        if results["format"] == "ico":
            try:
                sizes = img.info.get("sizes", [])
                sizes = [s[0] for s in sizes]

                for req in DPI_REQUIRED:
                    if req not in sizes:
                        results["dpi_missing"].append(req)

                for s in sizes:
                    if s <= 0 or s > 1024:
                        results["dpi_invalid"].append(s)

            except:
                results["errors"].append("ico_parse_error")

        else:
            # PNG/BMP must be at least 256x256
            if img.width < 256 or img.height < 256:
                results["dpi_invalid"].append(f"{img.width}x{img.height}")

        # --- D. Style detection ---
        results["style"] = self.detect_style(img)

        return results

    def detect_style(self, img):
        arr = np.array(img.convert("RGBA"))
        pixels = arr.reshape(-1, 4)
        pixels = pixels[pixels[:, 3] > 10]

        if len(pixels) == 0:
            return "unknown"

        brightness = np.mean(pixels[:, :3])
        max_rgb = np.max(pixels[:, :3], axis=1)
        min_rgb = np.min(pixels[:, :3], axis=1)
        saturation = np.mean(max_rgb - min_rgb)
        unique_colors = len(np.unique(pixels[:, :3], axis=0))

        if unique_colors < 8 and saturation < 20:
            return "monochrome"

        if brightness > 150 and saturation < 60:
            return "fluent"

        return "classic"

    def detect_duplicates(self, library):
        """
        Detect duplicate icons in the library.
        Returns list of pairs (id_a, id_b).
        """
        duplicates = []
        seen = {}

        for e in library.entries:
            if e.hash in seen:
                duplicates.append((seen[e.hash], e.id))
            else:
                seen[e.hash] = e.id

        return duplicates

import os
from PIL import Image
import base64
import json
import requests

DPI_SIZES = [16, 24, 32, 48, 64, 128, 256]


class AIIconGenerator:
    def __init__(self, api_url=None, api_key=None):
        self.api_url = api_url
        self.api_key = api_key

    def generate_png(self, prompt, out_path):
        """
        Generate PNG from text prompt using external AI backend.
        This is a generic wrapper. Replace with your backend.
        """

        if not self.api_url:
            raise Exception("AI backend URL not configured")

        payload = {
            "prompt": prompt,
            "size": 512
        }

        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {self.api_key}"
        }

        r = requests.post(self.api_url, headers=headers, data=json.dumps(payload))
        if r.status_code != 200:
            raise Exception("AI backend error")

        data = r.json()

        # Expect base64 PNG
        img_bytes = base64.b64decode(data["image"])
        with open(out_path, "wb") as f:
            f.write(img_bytes)

        return out_path

    def png_to_ico(self, png_path, ico_path):
        img = Image.open(png_path).convert("RGBA")

        icons = []
        for size in DPI_SIZES:
            resized = img.resize((size, size), Image.LANCZOS)
            icons.append(resized)

        icons[0].save(
            ico_path,
            format="ICO",
            sizes=[(s, s) for s in DPI_SIZES]
        )

        return ico_path

    def generate_icon(self, prompt, out_dir):
        """
        Full pipeline: prompt -> PNG -> ICO
        """
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        png_path = os.path.join(out_dir, "ai_icon.png")
        ico_path = os.path.join(out_dir, "ai_icon.ico")

        self.generate_png(prompt, png_path)
        self.png_to_ico(png_path, ico_path)

        return png_path, ico_path

python
import os
import json
import zipfile
import shutil


class IconPack:
    def __init__(self):
        pass

    def export_pack(self, library, history, out_path):
        """
        Export entire icon library + metadata into .iconpack (zip).
        """

        temp_dir = out_path + "_temp"
        if os.path.isdir(temp_dir):
            shutil.rmtree(temp_dir)
        os.makedirs(temp_dir)

        # --- icons ---
        icons_dir = os.path.join(temp_dir, "icons")
        os.makedirs(icons_dir)

        for e in library.entries:
            ext = os.path.splitext(e.path)[1].lower()
            out_icon = os.path.join(icons_dir, f"{e.id}{ext}")
            shutil.copyfile(e.path, out_icon)

        # --- metadata: tags + manifest ---
        metadata_dir = os.path.join(temp_dir, "metadata")
        os.makedirs(metadata_dir)

        tags = {
            str(e.id): e.tags
            for e in library.entries
        }

        with open(os.path.join(metadata_dir, "tags.json"), "w", encoding="utf-8") as f:
            json.dump(tags, f, indent=2)

        manifest = {
            "version": 1,
            "count": len(library.entries),
            "next_id": library.next_id,
            "icons": [
                {
                    "id": e.id,
                    "name": e.name,
                    "file": f"{e.id}{os.path.splitext(e.path)[1].lower()}",
                    "hash": e.hash,
                    "tags": e.tags
                }
                for e in library.entries
            ]
        }

        with open(os.path.join(metadata_dir, "manifest.json"), "w", encoding="utf-8") as f:
            json.dump(manifest, f, indent=2)

        # --- history ---
        with open(os.path.join(metadata_dir, "history.json"), "w", encoding="utf-8") as f:
            json.dump(history.snapshots, f, indent=2)

        # --- pack into zip ---
        with zipfile.ZipFile(out_path, "w", zipfile.ZIP_DEFLATED) as z:
            for root, dirs, files in os.walk(temp_dir):
                for file in files:
                    full = os.path.join(root, file)
                    rel = os.path.relpath(full, temp_dir)
                    z.write(full, rel)

        shutil.rmtree(temp_dir)
        return out_path

    def import_pack(self, pack_path, library, history):
        """
        Import .iconpack into library + history.
        """

        temp_dir = pack_path + "_temp"
        if os.path.isdir(temp_dir):
            shutil.rmtree(temp_dir)
        os.makedirs(temp_dir)

        # --- unpack ---
        with zipfile.ZipFile(pack_path, "r") as z:
            z.extractall(temp_dir)

        # --- load manifest ---
        manifest_path = os.path.join(temp_dir, "metadata", "manifest.json")
        with open(manifest_path, "r", encoding="utf-8") as f:
            manifest = json.load(f)

        # --- load tags ---
        tags_path = os.path.join(temp_dir, "metadata", "tags.json")
        with open(tags_path, "r", encoding="utf-8") as f:
            tags = json.load(f)

        # --- load history ---
        history_path = os.path.join(temp_dir, "metadata", "history.json")
        with open(history_path, "r", encoding="utf-8") as f:
            history.snapshots = json.load(f)

        # --- rebuild library ---
        library.entries = []
        library.next_id = manifest["next_id"]

        for item in manifest["icons"]:
            icon_file = os.path.join(temp_dir, "icons", item["file"])
            from iconlib.engine import IconEntry
            entry = IconEntry(
                item["id"],
                item["name"],
                icon_file,
                tags.get(str(item["id"]), [])
            )
            entry.hash = item["hash"]
            library.entries.append(entry)

        shutil.rmtree(temp_dir)
        return True

python
from PIL import Image, ImageEnhance, ImageFilter
import numpy as np
import os

DPI_SIZES = [16, 24, 32, 48, 64, 128, 256]


class IconNormalizer:
    def __init__(self):
        pass

    def normalize(self, src_path, out_path, target_style="fluent"):
        img = Image.open(src_path).convert("RGBA")

        # Step 1: normalize size
        img = self._normalize_size(img)

        # Step 2: normalize DPI
        dpi_icons = self._normalize_dpi(img)

        # Step 3: normalize style
        if target_style == "fluent":
            dpi_icons = [self._normalize_fluent(i) for i in dpi_icons]
        elif target_style == "classic":
            dpi_icons = [self._normalize_classic(i) for i in dpi_icons]
        elif target_style == "monochrome":
            dpi_icons = [self._normalize_monochrome(i) for i in dpi_icons]

        # Step 4: save ICO
        dpi_icons[0].save(
            out_path,
            format="ICO",
            sizes=[(s, s) for s in DPI_SIZES]
        )

        return out_path

    # --- SIZE NORMALIZATION ---

    def _normalize_size(self, img):
        w, h = img.size

        if w < 256 or h < 256:
            img = img.resize((256, 256), Image.LANCZOS)
            img = img.filter(ImageFilter.SMOOTH_MORE)

        if w > 512 or h > 512:
            img = img.resize((512, 512), Image.LANCZOS)

        return img

    # --- DPI NORMALIZATION ---

    def _normalize_dpi(self, img):
        icons = []
        for size in DPI_SIZES:
            resized = img.resize((size, size), Image.LANCZOS)
            icons.append(resized)
        return icons

    # --- STYLE NORMALIZATION ---

    def _normalize_fluent(self, img):
        img = ImageEnhance.Brightness(img).enhance(1.15)
        img = ImageEnhance.Color(img).enhance(0.65)
        img = ImageEnhance.Contrast(img).enhance(0.85)
        img = img.filter(ImageFilter.SMOOTH_MORE)
        return img

    def _normalize_classic(self, img):
        img = ImageEnhance.Contrast(img).enhance(1.4)
        img = ImageEnhance.Color(img).enhance(1.3)
        img = ImageEnhance.Sharpness(img).enhance(1.2)
        return img

    def _normalize_monochrome(self, img):
        arr = np.array(img)
        gray = np.mean(arr[:, :, :3], axis=2)
        mono = np.stack([gray, gray, gray, arr[:, :, 3]], axis=2)
        return Image.fromarray(mono.astype("uint8"), "RGBA")

import os
from iconlib.dll_import import DLLIconImporter
from iconlib.dll_export import DLLBuilder
from iconlib.validator import IconValidator
from iconlib.style_detector import IconStyleDetector


class IconMigrator:
    def __init__(self):
        pass

    def load_dll(self, path):
        importer = DLLIconImporter(path)
        icons = importer.import_icons()
        return icons

    def compare(self, dll_a_icons, dll_b_icons):
        """
        Compare two DLL icon sets.
        Returns dict with:
            - missing_in_b
            - missing_in_a
            - duplicates
            - changed
        """

        map_a = {i["id"]: i for i in dll_a_icons}
        map_b = {i["id"]: i for i in dll_b_icons}

        missing_in_b = [map_a[i] for i in map_a if i not in map_b]
        missing_in_a = [map_b[i] for i in map_b if i not in map_a]

        changed = []
        for i in map_a:
            if i in map_b:
                if map_a[i]["ico"] != map_b[i]["ico"]:
                    changed.append((map_a[i], map_b[i]))

        # detect duplicates by hash
        val = IconValidator()
        dup = []
        seen = {}
        for i in dll_a_icons + dll_b_icons:
            h = self._hash(i["ico"])
            if h in seen:
                dup.append((seen[h], i["id"]))
            else:
                seen[h] = i["id"]

        return {
            "missing_in_b": missing_in_b,
            "missing_in_a": missing_in_a,
            "changed": changed,
            "duplicates": dup
        }

    def migrate(self, dll_src_icons, dll_dst_path, out_path, ids_to_copy):
        """
        Copy selected icons from src DLL into new DLL.
        """

        builder = DLLBuilder()

        base_icon_id = 200
        group_id = 100

        # Add existing icons from destination DLL
        dst_importer = DLLIconImporter(dll_dst_path)
        dst_icons = dst_importer.import_icons()

        for item in dst_icons:
            builder.add_icon(item["ico_path"], group_id, base_icon_id)
            group_id += 1
            base_icon_id += 20

        # Add selected icons from source DLL
        for item in dll_src_icons:
            if item["id"] in ids_to_copy:
                temp_path = f"migrate_{item['id']}.ico"
                with open(temp_path, "wb") as f:
                    f.write(item["ico"])
                builder.add_icon(temp_path, group_id, base_icon_id)
                group_id += 1
                base_icon_id += 20

        builder.build(out_path)
        return out_path

    def _hash(self, data):
        import hashlib
        h = hashlib.sha256()
        h.update(data)
        return h.hexdigest()

import os
from PIL import Image
import hashlib
import numpy as np


class IconCache:
    def __init__(self):
        self.meta = {}        # id -> metadata
        self.thumb64 = {}     # id -> 64x64 thumbnail
        self.thumb128 = {}    # id -> 128x128 thumbnail
        self.full = {}        # id -> full image (lazy)
        self.index_name = {}  # name -> [ids]
        self.index_tag = {}   # tag -> [ids]
        self.index_style = {} # style -> [ids]
        self.index_hash = {}  # hash -> id
        self.index_dpi = {}   # dpi -> [ids]

    # --- BUILD CACHE FROM LIBRARY ---

    def build(self, library):
        from iconlib.style_detector import IconStyleDetector
        det = IconStyleDetector()

        for e in library.entries:
            # --- metadata ---
            meta = {
                "id": e.id,
                "name": e.name.lower(),
                "tags": [t.lower() for t in e.tags],
                "path": e.path,
                "hash": e.hash,
            }
            self.meta[e.id] = meta

            # --- style ---
            try:
                img = Image.open(e.path).convert("RGBA")
                style = det.detect_style(e.path)
            except:
                style = "unknown"

            meta["style"] = style

            # --- thumbnails ---
            try:
                self.thumb64[e.id] = img.resize((64, 64), Image.LANCZOS)
                self.thumb128[e.id] = img.resize((128, 128), Image.LANCZOS)
            except:
                pass

            # --- index: name ---
            self.index_name.setdefault(meta["name"], []).append(e.id)

            # --- index: tags ---
            for t in meta["tags"]:
                self.index_tag.setdefault(t, []).append(e.id)

            # --- index: style ---
            self.index_style.setdefault(style, []).append(e.id)

            # --- index: hash ---
            self.index_hash[e.hash] = e.id

            # --- index: DPI ---
            dpi = self._detect_dpi(e.path)
            for d in dpi:
                self.index_dpi.setdefault(d, []).append(e.id)

    # --- LAZY LOAD FULL IMAGE ---

    def get_full(self, id_):
        if id_ in self.full:
            return self.full[id_]

        path = self.meta[id_]["path"]
        img = Image.open(path).convert("RGBA")
        self.full[id_] = img
        return img

    # --- DPI DETECTION ---

    def _detect_dpi(self, path):
        try:
            img = Image.open(path)
            if img.format.lower() == "ico":
                sizes = img.info.get("sizes", [])
                return [s[0] for s in sizes]
            else:
                return [img.width]
        except:
            return []

    # --- SEARCH ENGINE ---

    def search(self, text):
        text = text.lower()

        results = set()

        # name match
        for name, ids in self.index_name.items():
            if text in name:
                results.update(ids)

        # tag match
        for tag, ids in self.index_tag.items():
            if text in tag:
                results.update(ids)

        # style match
        if text in self.index_style:
            results.update(self.index_style[text])

        return list(results)

    # --- FIND DUPLICATES ---

    def find_duplicates(self):
        dup = []
        seen = {}
        for id_, meta in self.meta.items():
            h = meta["hash"]
            if h in seen:
                dup.append((seen[h], id_))
            else:
                seen[h] = id_
        return dup

import os
import hashlib
from PIL import Image, ImageFilter, ImageEnhance
import numpy as np


class IconOptimizer:
    def __init__(self):
        pass

    # --- MAIN PIPELINE ---

    def optimize(self, src_path, out_path):
        img = Image.open(src_path).convert("RGBA")

        # Step 1: reduce noise
        img = img.filter(ImageFilter.SMOOTH_MORE)

        # Step 2: reduce saturation slightly
        img = ImageEnhance.Color(img).enhance(0.95)

        # Step 3: reduce contrast slightly
        img = ImageEnhance.Contrast(img).enhance(0.95)

        # Step 4: reduce alpha noise
        img = self._clean_alpha(img)

        # Step 5: reduce size if too large
        img = self._reduce_size(img)

        # Step 6: save optimized PNG
        img.save(out_path, optimize=True)

        return out_path

    # --- ALPHA CLEANING ---

    def _clean_alpha(self, img):
        arr = np.array(img)
        alpha = arr[:, :, 3]

        # remove tiny alpha noise
        alpha[alpha < 10] = 0

        arr[:, :, 3] = alpha
        return Image.fromarray(arr, "RGBA")

    # --- SIZE REDUCTION ---

    def _reduce_size(self, img):
        w, h = img.size
        if w > 512 or h > 512:
            img = img.resize((512, 512), Image.LANCZOS)
            img = img.filter(ImageFilter.SMOOTH)
        return img

    # --- DUPLICATE DETECTION ---

    def detect_duplicates(self, library):
        seen = {}
        dup = []

        for e in library.entries:
            h = self._hash_file(e.path)
            if h in seen:
                dup.append((seen[h], e.id))
            else:
                seen[h] = e.id

        return dup

    def _hash_file(self, path):
        h = hashlib.sha256()
        with open(path, "rb") as f:
            h.update(f.read())
        return h.hexdigest()

    # --- ICO OPTIMIZATION ---

    def optimize_ico(self, src_path, out_path):
        img = Image.open(src_path)

        if img.format.lower() != "ico":
            return self.optimize(src_path, out_path)

        sizes = img.info.get("sizes", [])
        valid_sizes = [s for s in sizes if s[0] <= 256]

        # rebuild ICO with only valid sizes
        layers = []
        for s in valid_sizes:
            layer = img.resize((s[0], s[0]), Image.LANCZOS)
            layer = layer.filter(ImageFilter.SMOOTH_MORE)
            layers.append(layer)

        layers[0].save(
            out_path,
            format="ICO",
            sizes=[(s[0], s[0]) for s in valid_sizes]
        )

        return out_path

import numpy as np
from PIL import Image
import base64
import json
import requests


class IconClassifier:
    def __init__(self, ai_url=None, ai_key=None):
        self.ai_url = ai_url
        self.ai_key = ai_key

    # --- MAIN ENTRY POINT ---

    def classify(self, path):
        img = Image.open(path).convert("RGBA")

        # Step 1: heuristic classification
        cat = self._heuristic(img)
        if cat != "unknown":
            return cat

        # Step 2: AI classification (optional)
        if self.ai_url:
            return self._ai_classify(path)

        return "unknown"

    # --- HEURISTIC CLASSIFIER ---

    def _heuristic(self, img):
        arr = np.array(img)
        h, w, _ = arr.shape

        # brightness
        brightness = np.mean(arr[:, :, :3])

        # saturation
        max_rgb = np.max(arr[:, :, :3], axis=2)
        min_rgb = np.min(arr[:, :, :3], axis=2)
        saturation = np.mean(max_rgb - min_rgb)

        # detect common shapes
        edges = self._edges(img)

        # folder-like (yellow-ish, rectangular)
        if self._is_folder(arr):
            return "folder"

        # document-like (white-ish rectangle)
        if self._is_document(arr):
            return "document"

        # warning triangle (yellow + strong edges)
        if self._is_warning(arr, edges):
            return "warning"

        # error (red circle)
        if self._is_error(arr):
            return "error"

        # checkmark (green + diagonal shape)
        if self._is_check(arr):
            return "check"

        # cloud shape (soft edges + white/blue)
        if self._is_cloud(arr):
            return "cloud"

        # gear shape (settings)
        if self._is_gear(edges):
            return "settings"

        # user silhouette
        if self._is_user(arr):
            return "user"

        return "unknown"

    # --- SHAPE DETECTORS ---

    def _edges(self, img):
        gray = img.convert("L")
        arr = np.array(gray)
        diff_h = np.abs(arr[:, 1:] - arr[:, :-1])
        diff_v = np.abs(arr[1:, :] - arr[:-1, :])
        return np.mean(diff_h) + np.mean(diff_v)

    def _is_folder(self, arr):
        # yellow-ish dominant color
        r, g, b = np.mean(arr[:, :, 0]), np.mean(arr[:, :, 1]), np.mean(arr[:, :, 2])
        return r > 180 and g > 150 and b < 100

    def _is_document(self, arr):
        # mostly white
        return np.mean(arr[:, :, :3]) > 200

    def _is_warning(self, arr, edges):
        # yellow + strong edges
        r, g, b = np.mean(arr[:, :, 0]), np.mean(arr[:, :, 1]), np.mean(arr[:, :, 2])
        return r > 200 and g > 180 and b < 80 and edges > 0.25

    def _is_error(self, arr):
        # red dominant
        r, g, b = np.mean(arr[:, :, 0]), np.mean(arr[:, :, 1]), np.mean(arr[:, :, 2])
        return r > 180 and g < 80 and b < 80

    def _is_check(self, arr):
        # green dominant
        r, g, b = np.mean(arr[:, :, 0]), np.mean(arr[:, :, 1]), np.mean(arr[:, :, 2])
        return g > 160 and r < 120 and b < 120

    def _is_cloud(self, arr):
        # white/blue soft shape
        r, g, b = np.mean(arr[:, :, 0]), np.mean(arr[:, :, 1]), np.mean(arr[:, :, 2])
        return r > 180 and g > 180 and b > 200

    def _is_gear(self, edges):
        # gears have strong radial edges
        return edges > 0.35

    def _is_user(self, arr):
        # silhouette detection (dark blob)
        brightness = np.mean(arr[:, :, :3])
        return brightness < 120

    # --- AI CLASSIFIER ---

    def _ai_classify(self, path):
        with open(path, "rb") as f:
            img_bytes = f.read()

        payload = {
            "image": base64.b64encode(img_bytes).decode("ascii")
        }

        headers = {
            "Content-Type": "application/json",
            "Authorization": f"Bearer {self.ai_key}"
        }

        r = requests.post(self.ai_url, headers=headers, data=json.dumps(payload))
        if r.status_code != 200:
            return "unknown"

        data = r.json()
        return data.get("category", "unknown")

import os
import numpy as np
from PIL import Image, ImageFilter, ImageEnhance

from iconlib.ai_generator import AIIconGenerator
from iconlib.normalizer import IconNormalizer
from iconlib.optimizer import IconOptimizer


class Fluent3DGenerator:
    def __init__(self, ai_url=None, ai_key=None):
        self.ai_url = ai_url
        self.ai_key = ai_key

    # --- MAIN PIPELINE ---

    def generate(self, prompt, out_dir):
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        # Step 1: AI base render
        base_png = os.path.join(out_dir, "base.png")
        ai = AIIconGenerator(api_url=self.ai_url, api_key=self.ai_key)
        ai.generate_png(f"{prompt}, fluent 3d icon, soft shadows, glossy highlights", base_png)

        # Step 2: apply 3D shading
        shaded_png = os.path.join(out_dir, "shaded.png")
        self._apply_3d_shading(base_png, shaded_png)

        # Step 3: normalize to Fluent 3D spec
        norm_png = os.path.join(out_dir, "norm.png")
        self._normalize_fluent3d(shaded_png, norm_png)

        # Step 4: DPI + ICO
        norm = IconNormalizer()
        ico_path = os.path.join(out_dir, "fluent3d.ico")
        norm.normalize(norm_png, ico_path, "fluent")

        # Step 5: optimize
        opt = IconOptimizer()
        opt_path = os.path.join(out_dir, "fluent3d_opt.ico")
        opt.optimize_ico(ico_path, opt_path)

        return {
            "base": base_png,
            "shaded": shaded_png,
            "normalized": norm_png,
            "ico": opt_path
        }

    # --- 3D SHADING ---

    def _apply_3d_shading(self, src, dst):
        img = Image.open(src).convert("RGBA")

        # soft shadow
        shadow = img.filter(ImageFilter.GaussianBlur(12))

        # blend shadow
        arr = np.array(img)
        sh = np.array(shadow)
        arr[:, :, :3] = (arr[:, :, :3] * 0.85 + sh[:, :, :3] * 0.15).astype("uint8")

        # glossy highlight
        gloss = img.filter(ImageFilter.GaussianBlur(4))
        gloss_arr = np.array(gloss)
        arr[:, :, :3] = (arr[:, :, :3] * 0.9 + gloss_arr[:, :, :3] * 0.1).astype("uint8")

        # ambient occlusion (AO)
        ao = img.filter(ImageFilter.EDGE_ENHANCE_MORE)
        ao_arr = np.array(ao)
        arr[:, :, :3] = (arr[:, :, :3] * 0.95 + ao_arr[:, :, :3] * 0.05).astype("uint8")

        Image.fromarray(arr, "RGBA").save(dst, optimize=True)

    # --- FLUENT 3D NORMALIZATION ---

    def _normalize_fluent3d(self, src, dst):
        img = Image.open(src).convert("RGBA")

        # pastel colors
        img = ImageEnhance.Color(img).enhance(0.75)

        # soft contrast
        img = ImageEnhance.Contrast(img).enhance(0.9)

        # soft brightness
        img = ImageEnhance.Brightness(img).enhance(1.1)

        # bloom
        bloom = img.filter(ImageFilter.GaussianBlur(6))
        arr = np.array(img)
        bl = np.array(bloom)
        arr[:, :, :3] = (arr[:, :, :3] * 0.85 + bl[:, :, :3] * 0.15).astype("uint8")

        Image.fromarray(arr, "RGBA").save(dst, optimize=True)

import os

import json


class IconLayoutEngine:
    def __init__(self):
        pass

    # --- MAIN ENTRY POINT ---

    def layout(self, icons, layout_type, cache, out_dir, spacing=16, padding=24):
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        if layout_type == "grid":
            return self._layout_grid(icons, cache, out_dir, spacing, padding)

        if layout_type == "toolbar":
            return self._layout_toolbar(icons, cache, out_dir, spacing, padding)

        if layout_type == "ribbon":
            return self._layout_ribbon(icons, cache, out_dir, spacing, padding)

        if layout_type == "sidebar":
            return self._layout_sidebar(icons, cache, out_dir, spacing, padding)

        if layout_type == "menu":
            return self._layout_menu(icons, cache, out_dir, spacing, padding)

        raise Exception("Unknown layout type")

    # --- GRID LAYOUT ---

    def _layout_grid(self, icons, cache, out_dir, spacing, padding):
        cols = 6
        rows = (len(icons) + cols - 1) // cols

        cell = 128
        w = cols * cell + (cols + 1) * spacing + padding * 2
        h = rows * cell + (rows + 1) * spacing + padding * 2

        img = Image.new("RGBA", (w, h), (255, 255, 255, 0))
        draw = ImageDraw.Draw(img)

        layout = []

        x = padding + spacing
        y = padding + spacing

        for i, id_ in enumerate(icons):
            thumb = cache.thumb128[id_]
            img.paste(thumb, (x, y), thumb)

            layout.append({
                "id": id_,
                "x": x,
                "y": y,
                "w": 128,
                "h": 128
            })

            x += cell + spacing
            if (i + 1) % cols == 0:
                x = padding + spacing
                y += cell + spacing

        preview = os.path.join(out_dir, "grid.png")
        img.save(preview, optimize=True)

        meta = os.path.join(out_dir, "grid.json")
        with open(meta, "w") as f:
            json.dump(layout, f, indent=2)

        return {"preview": preview, "meta": meta}

import numpy as np
from PIL import Image


class IconStyleAnalysis:
    def __init__(self):
        pass

    def analyze(self, path):
        img = Image.open(path).convert("RGBA")
        arr = np.array(img)

        # remove transparent pixels
        px = arr.reshape(-1, 4)
        px = px[px[:, 3] > 10]

        # --- COLOR METRICS ---
        unique_colors = len(np.unique(px[:, :3], axis=0))
        brightness = float(np.mean(px[:, :3]))
        max_rgb = np.max(px[:, :3], axis=1)
        min_rgb = np.min(px[:, :3], axis=1)
        saturation = float(np.mean(max_rgb - min_rgb))

        # --- OUTLINE DETECTION ---
        outline = self._detect_outline(arr)

        # --- GRADIENT DETECTION ---
        gradient = self._detect_gradient(arr)

        # --- SHADING DETECTION ---
        shading = self._detect_shading(arr)

        # --- STYLE CLASSIFICATION ---
        style = self._classify_style(unique_colors, saturation, gradient, shading, outline)

        return {
            "unique_colors": unique_colors,
            "brightness": brightness,
            "saturation": saturation,
            "outline": outline,
            "gradient": gradient,
            "shading": shading,
            "style": style
        }

    # --- OUTLINE DETECTION ---

    def _detect_outline(self, arr):
        gray = np.mean(arr[:, :, :3], axis=2)
        edges = np.abs(gray[:, 1:] - gray[:, :-1]).mean() + np.abs(gray[1:, :] - gray[:-1, :]).mean()
        return edges > 20

    # --- GRADIENT DETECTION ---

    def _detect_gradient(self, arr):
        h, w, _ = arr.shape
        left = np.mean(arr[:, :w//2, :3])
        right = np.mean(arr[:, w//2:, :3])
        diff = abs(left - right)
        return diff > 15

    # --- SHADING DETECTION (3D) ---

    def _detect_shading(self, arr):
        blur = np.mean(arr[:, :, :3]) - np.mean(arr[:, :, :3].astype(float) * 0.9)
        return blur > 5

    # --- STYLE CLASSIFICATION ---

    def _classify_style(self, unique_colors, saturation, gradient, shading, outline):
        # Fluent 3D
        if shading and gradient and unique_colors > 20:
            return "fluent3d"

        # Fluent 2D
        if gradient and saturation < 60 and unique_colors < 20:
            return "fluent2d"

        # Skeuomorphic
        if shading and unique_colors > 50:
            return "skeuomorphic"

        # Classic
        if unique_colors > 30 and saturation > 80:
            return "classic"

        # Monochrome
        if unique_colors < 5 and saturation < 20:
            return "monochrome"

        # Outline
        if outline and unique_colors < 10:
            return "outline"

        # Glyph
        if unique_colors < 8 and saturation < 40:
            return "glyph"

        return "unknown"

import numpy as np
from PIL import Image, ImageFilter, ImageEnhance


class IconTransform:
    def __init__(self):
        pass

    # --- MAIN ENTRY POINT ---

    def transform(self, src_path, out_path, mode="extrude"):
        img = Image.open(src_path).convert("RGBA")

        if mode == "extrude":
            out = self._extrude(img)
        elif mode == "bevel":
            out = self._bevel(img)
        elif mode == "emboss":
            out = self._emboss(img)
        elif mode == "glow":
            out = self._glow(img)
        else:
            raise Exception("Unknown transform mode")

        out.save(out_path, optimize=True)
        return out_path

    # --- EXTRUDE (pseudo‑3D depth) ---

    def _extrude(self, img):
        arr = np.array(img)
        shadow = img.filter(ImageFilter.GaussianBlur(8))
        sh = np.array(shadow)

        # shift shadow down-right
        extr = np.zeros_like(arr)
        extr[8:, 8:, :3] = sh[:-8, :-8, :3]
        extr[:, :, 3] = arr[:, :, 3]

        # blend
        arr[:, :, :3] = (arr[:, :, :3] * 0.85 + extr[:, :, :3] * 0.15).astype("uint8")
        return Image.fromarray(arr, "RGBA")

    # --- BEVEL (3D edges) ---

    def _bevel(self, img):
        arr = np.array(img)
        blur = img.filter(ImageFilter.GaussianBlur(4))
        sharp = img.filter(ImageFilter.EDGE_ENHANCE_MORE)

        bl = np.array(blur)
        sh = np.array(sharp)

        arr[:, :, :3] = (arr[:, :, :3] * 0.7 + bl[:, :, :3] * 0.2 + sh[:, :, :3] * 0.1).astype("uint8")
        return Image.fromarray(arr, "RGBA")

    # --- EMBOSS (relief) ---

    def _emboss(self, img):
        emb = img.filter(ImageFilter.EMBOSS)
        arr = np.array(emb)
        arr[:, :, 3] = np.array(img)[:, :, 3]
        return Image.fromarray(arr, "RGBA")

    # --- GLOW (soft halo) ---

    def _glow(self, img):
        glow = img.filter(ImageFilter.GaussianBlur(12))
        arr = np.array(img)
        gl = np.array(glow)

        arr[:, :, :3] = (arr[:, :, :3] * 0.8 + gl[:, :, :3] * 0.2).astype("uint8")
        return Image.fromarray(arr, "RGBA")

import os
import numpy as np
from PIL import Image, ImageFilter, ImageEnhance
import imageio

from iconlib.ai_generator import AIIconGenerator
from iconlib.fluent3d import Fluent3DGenerator


class Fluent3DAnimation:
    def __init__(self, ai_url=None, ai_key=None):
        self.ai_url = ai_url
        self.ai_key = ai_key

    # --- MAIN PIPELINE ---

    def animate(self, prompt, out_dir, frames=12, motion="rotate"):
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        ai = AIIconGenerator(api_url=self.ai_url, api_key=self.ai_key)
        f3d = Fluent3DGenerator(ai_url=self.ai_url, ai_key=self.ai_key)

        frame_paths = []

        # Step 1: generate frames
        for i in range(frames):
            angle = int((i / frames) * 360)

            motion_prompt = f"{prompt}, fluent 3d icon, {motion} {angle} degrees, soft shadows, glossy highlights"
            png_path = os.path.join(out_dir, f"frame_{i}.png")

            ai.generate_png(motion_prompt, png_path)

            # Step 2: apply Fluent 3D shading
            shaded_path = os.path.join(out_dir, f"frame_{i}_shaded.png")
            f3d._apply_3d_shading(png_path, shaded_path)

            # Step 3: normalize Fluent 3D style
            norm_path = os.path.join(out_dir, f"frame_{i}_norm.png")
            f3d._normalize_fluent3d(shaded_path, norm_path)

            frame_paths.append(norm_path)

        # Step 4: build GIF
        gif_path = os.path.join(out_dir, "animation.gif")
        frames_img = [Image.open(p) for p in frame_paths]
        frames_img[0].save(
            gif_path,
            save_all=True,
            append_images=frames_img[1:],
            duration=80,
            loop=0,
            optimize=True
        )

        # Step 5: build WebP
        webp_path = os.path.join(out_dir, "animation.webp")
        imageio.mimsave(webp_path, [np.array(Image.open(p)) for p in frame_paths], format="WEBP", fps=12)

        # Step 6: sprite sheet
        sprite_path = os.path.join(out_dir, "sprite.png")
        self._sprite_sheet(frame_paths, sprite_path)

        # Step 7: timeline metadata
        meta_path = os.path.join(out_dir, "timeline.json")
        self._timeline(meta_path, frames)

        return {
            "gif": gif_path,
            "webp": webp_path,
            "sprite": sprite_path,
            "timeline": meta_path
        }

    # --- SPRITE SHEET ---

    def _sprite_sheet(self, frame_paths, out_path):
        frames = [Image.open(p).convert("RGBA") for p in frame_paths]
        w, h = frames[0].size
        sheet = Image.new("RGBA", (w * len(frames), h))

        x = 0
        for f in frames:
            sheet.paste(f, (x, 0), f)
            x += w

        sheet.save(out_path, optimize=True)

    # --- TIMELINE METADATA ---

    def _timeline(self, out_path, frames):
        import json
        data = {
            "frames": frames,
            "fps": 12,
            "loop": True,
            "motion": "fluent3d"
        }
        with open(out_path, "w") as f:
            json.dump(data, f, indent=2)

import json
from PIL import Image, ImageDraw, ImageFont

from iconlib.color_analysis import IconColorAnalysis
from iconlib.layout import IconLayoutEngine


class UIAssetPipeline:
    def __init__(self, cache):
        self.cache = cache

    # --- MAIN ENTRY POINT ---

    def build(self, id_, ui_type, out_dir):
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        # Step 1: analyze icon
        mapper = IconUIMapper()
        color = IconColorAnalysis()

        icon_meta = mapper.map_action(self.cache.meta[id_]["path"])
        color_meta = color.analyze(self.cache.meta[id_]["path"])

        # Step 2: generate UI element
        if ui_type == "button":
            return self._button(id_, icon_meta, color_meta, out_dir)

        if ui_type == "toggle":
            return self._toggle(id_, icon_meta, color_meta, out_dir)

        if ui_type == "menu":
            return self._menu(id_, icon_meta, color_meta, out_dir)

        if ui_type == "toolbar":
            return self._toolbar(id_, icon_meta, color_meta, out_dir)

        raise Exception("Unknown UI type")

    # --- BUTTON ---

    def _button(self, id_, icon_meta, color_meta, out_dir):
        thumb = self.cache.thumb64[id_]
        w, h = 200, 64

        img = Image.new("RGBA", (w, h), (255, 255, 255, 0))
        draw = ImageDraw.Draw(img)

        # background
        bg = tuple(int(c) for c in color_meta["palette"][0])
        draw.rounded_rectangle((0, 0, w, h), radius=12, fill=bg)

        # icon
        img.paste(thumb, (16, (h - 64) // 2), thumb)

        # text
        font = ImageFont.load_default()
        draw.text((90, 22), icon_meta["action"], fill=(0, 0, 0), font=font)

        preview = os.path.join(out_dir, "button.png")
        img.save(preview, optimize=True)

        meta = {
            "type": "button",
            "action": icon_meta["action"],
            "color": color_meta,
            "icon": id_
        }

        with open(os.path.join(out_dir, "button.json"), "w") as f:
            json.dump(meta, f, indent=2)

        return {"preview": preview, "meta": os.path.join(out_dir, "button.json")}

    # --- TOGGLE ---

    def _toggle(self, id_, icon_meta, color_meta, out_dir):

        draw.rounded_rectangle((0, 0, w, h), radius=12, fill=(230, 230, 230))

        # toggle circle
        draw.ellipse((8, 8, 56, 56), fill=(255, 255, 255))

        # icon
        img.paste(thumb, (120, (h - 64) // 2), thumb)

        meta = {

        }

        return {"preview": preview, "meta": os.path.join(out_dir, "toggle.json")}

    # --- MENU ITEM ---

    def _menu(self, id_, icon_meta, color_meta, out_dir):

import numpy as np
from PIL import Image, ImageEnhance, ImageFilter

from iconlib.style_analysis import IconStyleAnalysis
from iconlib.normalizer import IconNormalizer
from iconlib.optimizer import IconOptimizer


class IconStyleConverter:
    def __init__(self):
        self.analyzer = IconStyleAnalysis()

    # --- MAIN ENTRY POINT ---

    def convert(self, src_path, out_dir, target_style="fluent"):
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        # Step 1: analyze original
        original_style = self.analyzer.analyze(src_path)

        # Step 2: load image
        img = Image.open(src_path).convert("RGBA")

        # Step 3: convert style
        if target_style == "fluent":
            out = self._to_fluent(img)
        elif target_style == "monochrome":
            out = self._to_monochrome(img)
        elif target_style == "flat":
            out = self._to_flat(img)
        else:
            raise Exception("Unknown target style")

        # Step 4: save PNG
        png_path = os.path.join(out_dir, "converted.png")
        out.save(png_path, optimize=True)

        # Step 5: normalize DPI + ICO
        norm = IconNormalizer()
        ico_path = os.path.join(out_dir, "converted.ico")
        norm.normalize(png_path, ico_path, target_style)

        # Step 6: optimize
        opt = IconOptimizer()
        opt_path = os.path.join(out_dir, "converted_opt.ico")
        opt.optimize_ico(ico_path, opt_path)

        # Step 7: analyze new style
        new_style = self.analyzer.analyze(opt_path)

        return {
            "png": png_path,
            "ico": opt_path,
            "original_style": original_style,
            "new_style": new_style
        }

    # --- CONVERSION: CLASSIC → FLUENT ---

    def _to_fluent(self, img):
        # pastel colors
        img = ImageEnhance.Color(img).enhance(0.75)

        # soft brightness
        img = ImageEnhance.Brightness(img).enhance(1.1)

        # soft contrast
        img = ImageEnhance.Contrast(img).enhance(0.9)

        # soft bloom
        bloom = img.filter(ImageFilter.GaussianBlur(6))
        arr = np.array(img)
        bl = np.array(bloom)
        arr[:, :, :3] = (arr[:, :, :3] * 0.85 + bl[:, :, :3] * 0.15).astype("uint8")

        return Image.fromarray(arr, "RGBA")

    # --- CONVERSION: ANY → MONOCHROME ---

    def _to_monochrome(self, img):
        gray = img.convert("L")
        arr = np.array(gray)
        rgba = np.zeros((arr.shape[0], arr.shape[1], 4), dtype=np.uint8)
        rgba[:, :, 0] = arr
        rgba[:, :, 1] = arr
        rgba[:, :, 2] = arr
        rgba[:, :, 3] = np.array(img)[:, :, 3]
        return Image.fromarray(rgba, "RGBA")

    # --- CONVERSION: SKEUOMORPHIC → FLAT ---

    def _to_flat(self, img):
        # remove shading
        flat = img.filter(ImageFilter.SMOOTH_MORE)

        # remove gradients
        flat = ImageEnhance.Color(flat).enhance(0.5)

        # increase brightness
        flat = ImageEnhance.Brightness(flat).enhance(1.2)

        # reduce contrast
        flat = ImageEnhance.Contrast(flat).enhance(0.8)

        return flat

import os
from PIL import Image, ImageDraw, ImageFont, ImageFilter, ImageEnhance

from iconlib.normalizer import IconNormalizer
from iconlib.optimizer import IconOptimizer


class TextIconGenerator:
    def __init__(self):
        pass

    # --- MAIN ENTRY POINT ---

    def generate(self, text, out_dir, style="fluent", size=256):
        if not os.path.isdir(out_dir):
            os.makedirs(out_dir)

        # Step 1: glyph → raster
        png_path = os.path.join(out_dir, "glyph.png")
        self._glyph_to_png(text, png_path, size, style)

        # Step 2: style transform
        styled_path = os.path.join(out_dir, "styled.png")
        self._apply_style(png_path, styled_path, style)

        # Step 3: normalize DPI + ICO
        norm = IconNormalizer()
        ico_path = os.path.join(out_dir, "glyph.ico")
        norm.normalize(styled_path, ico_path, style)

        # Step 4: optimize
        opt = IconOptimizer()
        opt_path = os.path.join(out_dir, "glyph_opt.ico")
        opt.optimize_ico(ico_path, opt_path)

        return {
            "png": styled_path,
            "ico": opt_path
        }

    # --- TEXT → GLYPH PNG ---

    def _glyph_to_png(self, text, out_path, size, style):
        img = Image.new("RGBA", (size, size), (255, 255, 255, 0))
        draw = ImageDraw.Draw(img)

        font = ImageFont.load_default()

        w, h = draw.textsize(text, font=font)
        x = (size - w) // 2
        y = (size - h) // 2

        draw.text((x, y), text, fill=(0, 0, 0), font=font)
        img.save(out_path, optimize=True)

    # --- STYLE APPLICATION ---

    def _apply_style(self, src, dst, style):
        img = Image.open(src).convert("RGBA")

        if style == "fluent":
            img = ImageEnhance.Color(img).enhance(0.75)
            img = ImageEnhance.Brightness(img).enhance(1.1)
            img = ImageEnhance.Contrast(img).enhance(0.9)
            bloom = img.filter(ImageFilter.GaussianBlur(6))
            arr = np.array(img)
            bl = np.array(bloom)
            arr[:, :, :3] = (arr[:, :, :3] * 0.85 + bl[:, :, :3] * 0.15).astype("uint8")
            Image.fromarray(arr, "RGBA").save(dst, optimize=True)
            return

        if style == "monochrome":
            gray = img.convert("L")
            arr = np.array(gray)
            rgba = np.zeros((arr.shape[0], arr.shape[1], 4), dtype=np.uint8)
            rgba[:, :, 0] = arr
            rgba[:, :, 1] = arr
            rgba[:, :, 2] = arr
            rgba[:, :, 3] = np.array(img)[:, :, 3]
            Image.fromarray(rgba, "RGBA").save(dst, optimize=True)
            return

        if style == "outline":
            outline = img.filter(ImageFilter.FIND_EDGES)
            arr = np.array(outline)
            arr[:, :, 3] = np.array(img)[:, :, 3]
            Image.fromarray(arr, "RGBA").save(dst, optimize=True)
            return

        # classic
        img = ImageEnhance.Color(img).enhance(1.4)
        img = ImageEnhance.Contrast(img).enhance(1.2)
        img.save(dst, optimize=True)

def text_icon(self, text, out_dir, style="fluent", size=256):
    from iconlib.text_icon import TextIconGenerator
    gen = TextIconGenerator()
    return gen.generate(text, out_dir, style, size)

def text_icon_gui(self):
    text = tk.simpledialog.askstring("Text", "Enter text:")
    style = tk.simpledialog.askstring("Style", "fluent/monochrome/outline/classic")
    out_dir = filedialog.askdirectory(title="Output directory")

    result = self.iconstudio.text_icon(text, out_dir, style)
    messagebox.showinfo("OK", f"Icon generated:\n{result['ico']}")
